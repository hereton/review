const devBypass = import.meta.env.VITE_DEV_BYPASS === 'true'
const devToken = import.meta.env.VITE_DEV_TOKEN as string | undefined

function getAuthHeader(): Record<string, string> {
  let token: string | undefined
  if (devBypass) {
    token = devToken || ''
  } else {
    token = useAuthStore.getState().token || undefined
  }
  return token !== undefined ? { Authorization: `Bearer ${token}` } : {}
}

▶ Anti-patterns 3
High No TanStack Query — manual useState/useEffect data fetching

All 39 hooks manage loading, error, and data state manually with useState + useEffect. No caching, no background refetching, no request deduplication, no stale-while-revalidate. Every component remount triggers a fresh API call.

export function useApplications(search: string, sortBy: string, sortOrder: 'asc'|'desc') {
  const [apps, setApps] = useState<App[]>([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string|null>(null)

  useEffect(() => {
    setLoading(true)
    api.get<AppListResponse>('/api/v1/applications?...')
      .then(res => { setApps(res.data); setLoading(false) })
      .catch(err => { setError(err.message); setLoading(false) })
  }, [search, sortBy, sortOrder])

  return { apps, loading, error, refetch: () => {} }
}
High CRUD hook duplication across 39 hooks

Every data hook repeats the same pattern: useState for data/loading/error + useEffect for fetch + refetch function. 39 hooks with near-identical boilerplate. A generic factory or TanStack Query would eliminate ~60% of hook code.

Medium No query cancellation or abort

No AbortController usage in fetch hooks. Navigating away from a page doesn't cancel pending requests. Race conditions possible with rapid filter/sort changes.

▶ Healthy Patterns 4
Good hook naming convention — useApplications, useRoles, usePermissions. Clear, descriptive names following React convention.
Comprehensive domain coverage — 39 hooks covering all API endpoints: applications, roles, permissions, employees, delegations, OU mappings, ABAC attributes/groups, audit logs, PC users. No missing data hooks.
Separation of concerns — Business logic in hooks, rendering in components. Each hook encapsulates API call + state management. Good encapsulation.
Virtual scrolling support — @tanstack/react-virtual for large data sets (employee directory). Performance-conscious design.
